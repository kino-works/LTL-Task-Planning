(define (domain housework)
    (:requirements :strips :typing :negative-preconditions)
    (:types agent room item appliance desk container)
    (:predicates
        (agent_at ?a - agent ?r - room)
        (agent_hand_free ?a - agent)
        (agent_has_item ?a - agent ?i - item)

        (at ?x - (either item desk) ?loc - (either room appliance desk container))

        (ap_at ?ap - appliance ?r - room)
        (on ?ap - appliance)

        (is_dishcloth ?i - item)
        (is_phone ?i - item)
        (is_dish ?i - item)

        (is_charger ?ap - appliance)

        (is_shelf ?c - container)
        (is_desk ?d - desk)

        (wiped ?d - desk)
        (charged ?i - item)
    )
    (:action goto
        :parameters (?a - agent ?from - room ?to - room)
        :precondition (agent_at ?a ?from)
        :effect (and
            (not (agent_at ?a ?from))
            (agent_at ?a ?to)
        )
    )
    (:action pick
        :parameters (?a - agent ?i - item ?loc - (either room appliance desk container))
        :precondition (and
            (agent_at ?a ?loc)
            (agent_hand_free ?a)
            (at ?i ?loc)
        )
        :effect (and
            (not (at ?i ?loc))
            (not (agent_hand_free ?a))
            (agent_has_item ?a ?i)
        )
    )
    (:action place
        :parameters (?a - agent ?i - item ?loc - (either room appliance desk container))
        :precondition (and
            (agent_at ?a ?loc)
            (agent_has_item ?a ?i)
            (not (at ?i ?loc))
        )
        :effect (and
            (at ?i ?loc)
            (agent_hand_free ?a)
            (not (agent_has_item ?a ?i))
        )
    )
    (:action turn_on
        :parameters (?a - agent ?ap - appliance ?r - room)
        :precondition (and
            (agent_at ?a ?r)
            (agent_hand_free ?a)
            (ap_at ?ap ?r)
            (not (on ?ap))
        )
        :effect (on ?ap)
    )
    (:action turn_off
        :parameters (?a - agent ?ap - appliance ?r - room)
        :precondition (and
            (agent_at ?a ?r)
            (agent_hand_free ?a)
            (ap_at ?ap ?r)
            (on ?ap)
        )
        :effect (not (on ?ap))
    )
    (:action wipe
      :parameters (?a - agent ?i - item ?r - room ?d - desk)
      :precondition (and
        (agent_at ?a ?r)
        (agent_has_item ?a ?i)
        (is_dishcloth ?i)
        (is_desk ?d)
      )
      :effect (wiped ?d)
    )
    (:action wait_charge_phone
        :parameters (?a - agent ?i - item ?ap - appliance ?r - room)
        :precondition (and
            (agent_at ?a ?r)
            (agent_hand_free ?a)
            (ap_at ?ap ?r)
            (at ?i ?ap)
            (on ?ap)
            (not (charged ?i))
            (is_phone ?i)
            (is_charger ?ap)
        )
        :effect (charged ?i)
    )
)

(define (problem doing_housework)
    (:domain housework)
    (:objects
        robot1 - agent
        kitchen bedroom livingroom - room
        charger - appliance
        dish_1 dish_2 dish_3 dishcloth phone - item
        desk - desk
        shelf - container
    )
    (:init
        (agent_at robot1 bedroom)
        (agent_hand_free robot1)

        (at dishcloth livingroom)
        (at phone bedroom)
        (at dish_1 kitchen)
        (at dish_2 kitchen)
        (at dish_3 kitchen)
        (at desk livingroom)
        (at shelf kitchen)

        (ap_at charger bedroom)
        (not (on charger))

        (not (wiped desk))
        (not (charged phone))

        (is_dishcloth dishcloth)
        (is_phone phone)
        (is_dish dish_1)
        (is_dish dish_2)
        (is_dish dish_3)

        (is_charger charger)

        (is_shelf shelf)
        (is_desk desk)
    )
    (:goal (and
        (wiped desk)
        (at dishcloth desk)
        (charged phone)
        (at phone desk)
        (at dish_1 shelf)
        (at dish_2 shelf)
        (at dish_3 shelf)
    ))
)
