I’m a service robot currently located in the bedroom of a house.
My goals are to heat the food, place the water bottle on the living room desk, and place the eggs in the egg container.
To accomplish each goal, I must first navigate to the room containing the target object.
Heating food requires using the microwave in the kitchen and waiting 3 minutes.
After heating, the food should be placed on the desk in the living room.
Placing the water bottle simply involves picking it up in the kitchen and carrying it to the living room desk.
Storing the eggs means picking them up from the kitchen and placing them into the egg container in the kitchen.
The robot can only use one hand at a time and can hold only one item.

Domain file:
(define (domain housework)
    (:requirements :strips :typing :negative-preconditions)
    (:types agent room item appliance desk)
    (:predicates
        (agent_at ?a - agent ?r - room)
        (agent_hand_free ?a - agent)
        (agent_has_item ?a - agent ?i - item)

        (at ?x - (either item desk) ?loc - (either room appliance desk))

        (ap_at ?ap - appliance ?r - room)
        (on ?ap - appliance)

        (is_food ?i - item)
        (is_water_bottle ?i - item)
        (is_eggs ?i - item)

        (is_microwave ?ap - appliance)
        (is_egg_container ?ap - appliance)

        (is_desk ?d - desk)

        (heated ?i - item)
    )
    (:action goto
        :parameters (?a - agent ?from - room ?to - room)
        :precondition (agent_at ?a ?from)
        :effect (and
            (not (agent_at ?a ?from))
            (agent_at ?a ?to)
        )
    )
    (:action pick
        :parameters (?a - agent ?i - item ?loc - (either room appliance desk))
        :precondition (and
            (agent_at ?a ?loc)
            (agent_hand_free ?a)
            (at ?i ?loc)
        )
        :effect (and
            (not (at ?i ?loc))
            (not (agent_hand_free ?a))
            (agent_has_item ?a ?i)
        )
    )
    (:action place
        :parameters (?a - agent ?i - item ?loc - (either room appliance desk))
        :precondition (and  
            (agent_at ?a ?loc)
            (agent_has_item ?a ?i)
            (not (at ?i ?loc))
        )
        :effect (and
            (at ?i ?loc)
            (agent_hand_free ?a)
            (not (agent_has_item ?a ?i))
        )
    )
    (:action turn_on
        :parameters (?a - agent ?ap - appliance ?r - room)
        :precondition (and
            (agent_at ?a ?r)
            (agent_hand_free ?a)
            (ap_at ?ap ?r)
            (not (on ?ap))
        )
        :effect (on ?ap)
    )
    (:action turn_off
        :parameters (?a - agent ?ap - appliance ?r - room)
        :precondition (and
            (agent_at ?a ?r)
            (agent_hand_free ?a)
            (ap_at ?ap ?r)
            (on ?ap)
        )
        :effect (not (on ?ap))
    )
    (:action wait_heat_food
        :parameters (?a - agent ?i - item ?ap - appliance ?r - room)
        :precondition (and
            (agent_at ?a ?r)
            (agent_hand_free ?a)
            (ap_at ?ap ?r)
            (at ?i ?ap)
            (on ?ap)
            (not (heated ?i))
            (is_food ?i)
        )
        :effect (and
            (heated ?i)
        )
    )
)

Problem file:
(define (problem doing_housework)
    (:domain housework)
    (:objects
        robot1 - agent
        kitchen bathroom bedroom livingroom - room
        microwave egg_container - appliance
        food water_bottle eggs - item
        desk - desk
    )
    (:init
        (agent_at robot1 bedroom)
        (agent_hand_free robot1)

        (at food kitchen)
        (at water_bottle kitchen)
        (at eggs kitchen)
        (at desk livingroom)

        (ap_at egg_container kitchen)
        (ap_at microwave kitchen)

        (not (on microwave))
        (not (on egg_container))

        (not (heated food))

        (is_food food)
        (is_water_bottle water_bottle)
        (is_eggs is_eggs)

        (is_microwave microwave)
        (is_egg_container egg_container)

        (is_desk desk)
    )
    (:goal (and
        (heated food)
        (at food desk)
        (at water_bottle desk)
        (at eggs egg_container)
    ))
)
