I’m a service robot currently located in the bedroom of a house.
My goals are to wipe the living room desk, charge the phone, and place the water bottle on the living room desk.
To wipe the desk, I need the dishcloth in the living room.
To charge the phone, I must use the charger in the bedroom and wait 10 minutes.
After charging, I’ll bring the phone to the living room desk.
To place the water bottle, I simply pick it up in the kitchen and carry it to the living room desk.

Domain file:
(define (domain housework)
    (:requirements :strips :typing)
    (:types agent room item appliance desk)
    (:predicates
        (agent_at ?a - agent ?r - room)
        (agent_hand_free ?a - agent)
        (agent_has_item ?a - agent ?i - item)

        (at ?i - item ?loc - (either room appliance desk))

        (ap_at ?ap - appliance ?r - room)
        (on ?ap - appliance)

        (clean_desk ?d - desk)
        (charged ?i - item)
    )
    (:action goto
        :parameters (?a - agent ?from - room ?to - room)
        :precondition (agent_at ?a ?from)
        :effect (and
            (not (agent_at ?a ?from))
            (agent_at ?a ?to)
        )
    )
    (:action pick
        :parameters (?a - agent ?i - item ?loc - (either room appliance desk))
        :precondition (and
            (agent_hand_free ?a)
            (at ?i ?loc)
        )
        :effect (and
            (not (at ?i ?loc))
            (not (agent_hand_free ?a))
            (agent_has_item ?a ?i)
        )
    )
    (:action place
        :parameters (?a - agent ?i - item ?loc - (either room appliance desk))
        :precondition (and
            (agent_has_item ?a ?i)
            (not (at ?i ?loc))
        )
        :effect (and
            (at ?i ?loc)
            (agent_hand_free ?a)
            (not (agent_has_item ?a ?i))
        )
    )
    (:action turn_on
        :parameters (?a - agent ?ap - appliance ?r - room)
        :precondition (and
            (agent_at ?a ?r)
            (ap_at ?ap ?r)
            (not (on ?ap))
        )
        :effect (on ?ap)
    )
    (:action turn_off
        :parameters (?a - agent ?ap - appliance ?r - room)
        :precondition (and
            (agent_at ?a ?r)
            (ap_at ?ap ?r)
            (on ?ap)
        )
        :effect (not (on ?ap))
    )
    (:action wipe
      :parameters (?a - agent ?i - item ?r - room ?d - desk)
      :precondition (and (agent_at ?a ?r) (at ?i ?r))
      :effect (clean_desk ?d)
    )
    (:action wait_charge_phone
        :parameters (?a - agent ?i - item ?ap - appliance ?r - room)
        :precondition (and
            (agent_at ?a ?r)
            (ap_at ?ap ?r)
            (at ?i ?ap)
            (on ?ap)
            (not (charged ?i))
        )
        :effect (and
            (charged ?i)
        )
    )
)

Problem file:
(define (problem doing_housework)
    (:domain housework)
    (:objects
        robot1 - agent
        kitchen bathroom bedroom livingroom - room
        charger - appliance
        dishcloth phone water_bottle - item
        desk - desk
    )
    (:init
        (agent_at robot1 bedroom)
        (agent_hand_free robot1)

        (at dishcloth livingroom)
        (at phone bedroom)
        (at water_bottle kitchen)
        (at desk livingroom)

        (ap_at charger bedroom)
        (not (on charger))
    )
    (:goal (and
        (clean_desk desk)
        (charged phone)
        (at water_bottle desk)
    ))
)
