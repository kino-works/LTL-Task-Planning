I’m a service robot currently located in the bedroom of a house.
My goals are to wipe the living room desk, charge the phone, and place the water bottle on the living room desk.
To wipe the desk, I need the dishcloth in the living room.
To charge the phone, I must use the charger in the bedroom and wait 10 minutes.
After charging, I’ll bring the phone to the living room desk.
To place the water bottle, I simply pick it up in the kitchen and carry it to the living room desk.

Domain file:
(define (domain housework)
  (:requirements :strips :typing :durative-actions)
  (:types robot room object appliance)
  
  (:predicates
    (at ?r - robot ?rm - room)
    (at ?o - object ?loc - (either room appliance))
    (on ?a - appliance)
    (wiped ?loc - room)
    (charged ?o - object)
  )
  
  (:action goto
    :parameters (?r - robot ?from ?to - room)
    :precondition (at ?r ?from)
    :effect (and (not (at ?r ?from)) (at ?r ?to))
  )
  
  (:action pick
    :parameters (?r - robot ?o - object ?loc - (either room appliance))
    :precondition (and (at ?r ?loc) (at ?o ?loc))
    :effect (not (at ?o ?loc))
  )
  
  (:action place
    :parameters (?r - robot ?o - object ?loc - (either room appliance))
    :precondition (at ?r ?loc)
    :effect (at ?o ?loc)
  )
  
  (:action turnOn
    :parameters (?r - robot ?a - appliance ?rm - room)
    :precondition (and (at ?r ?rm) (at ?a ?rm))
    :effect (on ?a)
  )
  
  (:action turnOff
    :parameters (?r - robot ?a - appliance ?rm - room)
    :precondition (and (at ?r ?rm) (at ?a ?rm) (on ?a))
    :effect (not (on ?a))
  )
  
  (:action wipe
    :parameters (?r - robot ?w - object ?loc - room)
    :precondition (and (at ?r ?loc) (at ?w ?loc))
    :effect (wiped ?loc)
  )
  
  (:durative-action wait
    :parameters (?r - robot ?o - object ?a - appliance ?loc - room)
    :condition (and (over all (at ?r ?loc))
                    (over all (at ?o ?loc))
                    (over all (on ?a)))
    :effect (at end (charged ?o))
  )
)

Problem file:
(define (problem doing_housework)
  (:domain housework)
  (:objects
    robot1 - robot
    kitchen bathroom bedroom livingroom - room
    charger desk - appliance
    dishcloth phone water_bottle - object
  )
  (:init
    (at robot1 bedroom)
    (at dishcloth livingroom)
    (at phone bedroom)
    (at charger bedroom)
    (at water_bottle kitchen)
    (at desk livingroom)
    (not (on charger))
  )
  (:goal (and
    (wiped desk)
    (charged phone)
    (at water_bottle desk)
  ))
)
