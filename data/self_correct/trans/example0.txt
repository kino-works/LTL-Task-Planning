I'm a service robot currently located in the bedroom of a house.
My task is to cook toast, boil water, and cook cup ramen, placing each finished item neatly on the living room desk.
I must first navigate to the room containing the target object.
Additionally, each cooking-related task requires waiting for a specific duration: 3 minutes for bread and cup ramen, and 5 minutes for boiling water.
After cooking, each prepared item should be placed on the desk in the living room to complete my goals.

Domain file:
(define (domain housework)
    (:requirements :strips :typing)
    (:types agent room item appliance desk)
    (:predicates
        (agent_at ?a - agent ?r - room)
        (agent_hand_free ?a - agent)
        (agent_has_item ?a - agent ?i - item)

        (at ?i - item ?loc - (either room appliance desk))

        (ap_at ?ap - appliance ?r - room)
        (on ?ap - appliance)

        (cooked ?i - item)
        (boiled ?i - item)
    )
    (:action goto
        :parameters (?a - agent ?from - room ?to - room)
        :precondition (agent_at ?a ?from)
        :effect (and
            (not (agent_at ?a ?from))
            (agent_at ?a ?to)
        )
    )
    (:action pick
        :parameters (?a - agent ?i - item ?loc - (either room appliance desk))
        :precondition (and
            (agent_hand_free ?a)
            (at ?i ?loc)
        )
        :effect (and
            (not (at ?i ?loc))
            (not (agent_hand_free ?a))
            (agent_has_item ?a ?i)
        )
    )
    (:action place
        :parameters (?a - agent ?i - item ?loc - (either room appliance desk))
        :precondition (and
            (agent_has_item ?a ?i)
            (not (at ?i ?loc))
        )
        :effect (and
            (at ?i ?loc)
            (agent_hand_free ?a)
            (not (agent_has_item ?a ?i))
        )
    )
    (:action turn_on
        :parameters (?a - agent ?ap - appliance ?r - room)
        :precondition (and
            (agent_at ?a ?r)
            (ap_at ?ap ?r)
            (not (on ?ap))
        )
        :effect (on ?ap)
    )
    (:action turn_off
        :parameters (?a - agent ?ap - appliance ?r - room)
        :precondition (and
            (agent_at ?a ?r)
            (ap_at ?ap ?r)
            (on ?ap)
        )
        :effect (not (on ?ap))
    )
    (:action wait_cook_bread
        :parameters (?a - agent ?i - item ?ap - appliance ?r - room)
        :precondition (and
            (agent_at ?a ?r)
            (ap_at ?ap ?r)
            (at ?i ?ap)
            (on ?ap)
            (not (toasted ?i))
        )
        :effect (and
            (toasted ?i)
        )
    )
    (:action wait_boil_water
        :parameters (?a - agent ?i - item ?ap - appliance ?r - room)
        :precondition (and
            (agent_at ?a ?r)
            (ap_at ?ap ?r)
            (at ?i ?ap)
            (on ?ap)
            (not (boiled ?i))
        )
        :effect (and
            (boiled ?i)
        )
    )
    (:action wait_cook_ramen
        :parameters (?a - agent ?i - item ?ap - appliance ?r - room)
        :precondition (and
            (agent_at ?a ?r)
            (ap_at ?ap ?r)
            (at ?i ?ap)
            (on ?ap)
            (not (cooked ?i))
        )
        :effect (and
            (cooked ?i)
        )
    )
)

Problem file:
(define (problem doing_housework)
    (:domain housework)
    (:objects
        robot1 - agent
        kitchen bathroom bedroom livingroom - room
        toaster stove water_dispenser - appliance
        bread cup_ramen kettle - item
        desk - desk
    )
    (:init
        (agent_at robot1 bedroom)
        (agent_hand_free robot1)

        (at bread kitchen)
        (at cup_ramen kitchen)
        (at kettle kitchen)
        (at desk livingroom)

        (ap_at toaster kitchen)
        (ap_at stove kitchen)
        (ap_at water_dispenser kitchen)

        (not (on toaster))
        (not (on stove))
        (not (on water_dispenser))
    )
    (:goal (and
        (cooked bread)
        (at bread desk)
        (boiled kettle)
        (at kettle desk)
        (cooked cup_ramen)
        (at cup_ramen desk)
    ))
)
